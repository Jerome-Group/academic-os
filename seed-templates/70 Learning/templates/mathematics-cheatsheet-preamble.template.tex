\newcommand{\KeepBlock}[1]{%
  \begingroup\par
  \ifnum\sheet@keepdepth=0\relax\SheetNeedspace{\SheetOpeningLines\baselineskip}\fi
  \advance\sheet@keepdepth by1\relax
  #1%
  \par\endgroup
}
\newcommand{\PassageHighlight}[2]{%
  \par\begingroup
  \setbox\sheet@highlightbox=\vtop{%
    \hsize=\linewidth
    \noindent{\color{SheetInk#1}#2\par}%
    \global\sheet@highlightdepth=\prevdepth
  }%
  \setlength{\fboxsep}{0pt}%
  \noindent\colorbox{SheetHighlight#1}{\box\sheet@highlightbox}\par
  % Preserve the leading from the last line of the passage to the next line.
  \prevdepth=\sheet@highlightdepth
  \endgroup
}
